# --- YORDAMCHI FUNKSIYALAR ---
def clean_markdown(text):
    if not text:
        return ""
    for char in ["_", "*", "`", "[", "]"]:
        text = str(text).replace(char, f"\\{char}")
    return text


def get_uzb_time():
    uzb_tz = timezone(timedelta(hours=5))
    return datetime.now(uzb_tz)


def safe_send_message(chat_id, text, **kwargs):
    try:
        return bot.send_message(chat_id, text, **kwargs)
    except Exception as e:
        print(f"Xabar yuborishda xatolik: {e}")
        return None


def load_tasks():
    if os.path.exists(TASKS_FILE):
        try:
            with open(TASKS_FILE, "r", encoding="utf-8") as f:
                content = f.read().strip()
                if not content:
                    return {}
                return json.loads(content)
        except Exception as e:
            print(f"Fayl o'qishda xatolik: {e}")
    return {}


def save_tasks(tasks):
    try:
        with open(TASKS_FILE, "w", encoding="utf-8") as f:
            json.dump(tasks, f, ensure_ascii=False, indent=4)
    except Exception as e:
        print(f"Fayl saqlashda xatolik: {e}")

def reminder_checker():
    while True:
        try:
            now_str = get_uzb_time().strftime("%Y-%m-%d %H:%M")
            tasks_data = load_tasks()
            updated = False

            for user_id, content in list(tasks_data.items()):
                user_tasks = (
                    content
                    if isinstance(content, list)
                    else content.get("tasks", [])
                )
                remaining_tasks = []

                for task in user_tasks:
                    if task.get("remind_at") and task["remind_at"] <= now_str:
                        res = safe_send_message(
                            int(user_id),
                            f"⏰ **ESLATMA!**\n\n📌 **Vazifa:** {clean_markdown(task['text'])}\n\nVaqti bo'ldi, bajarishni unutmang!",
                            parse_mode="Markdown",
                        )
                        if res:
                            updated = True
                        else:
                            remaining_tasks.append(task)
                    else:
                        remaining_tasks.append(task)

                if isinstance(content, list):
                    tasks_data[user_id] = remaining_tasks
                else:
                    tasks_data[user_id]["tasks"] = remaining_tasks

            if updated:
                save_tasks(tasks_data)
        except Exception as e:
            print(f"Eslatishda xatolik: {e}")

        time.sleep(10)


threading.Thread(target=reminder_checker, daemon=True).start()

@bot.callback_query_handler(func=lambda call: call.data.startswith("done_"))
def complete_task(call):
    user_id = str(call.message.chat.id)
    task_idx = int(call.data.split("_")[1])
    tasks_data = load_tasks()

    content = tasks_data.get(user_id, [])
    user_tasks = (
        content if isinstance(content, list) else content.get("tasks", [])
    )

    if 0 <= task_idx < len(user_tasks):
        completed = user_tasks.pop(task_idx)

        if isinstance(content, list):
            tasks_data[user_id] = user_tasks
        else:
            tasks_data[user_id]["tasks"] = user_tasks

        save_tasks(tasks_data)

        try:
            bot.answer_callback_query(call.id, "Vazifa bajarildi!")
            bot.edit_message_text(
                f"✅ **Bajarildi:** {clean_markdown(completed['text'])}",
                chat_id=call.message.chat.id,
                message_id=call.message.message_id,
                parse_mode="Markdown",
            )
        except Exception as e:
            print(f"Bajarildi belgilashda xatolik: {e}")

@bot.message_handler(func=lambda message: True)
def handle_text(message):
    if message.text == "➕ Vazifa qo'shish":
        safe_send_message(
            message.chat.id,
            "Yangi vazifangiz matnini yuboring (Masalan: *Bugun 21:50 da dars qilish kerak*):",
            parse_mode="Markdown",
        )
        return

    user_id = str(message.chat.id)
    user_full_name = message.from_user.first_name or "Foydalanuvchi"
    if message.from_user.username:
        user_full_name += f" (@{message.from_user.username})"

    task_text = message.text
    now = get_uzb_time()

    time_match = re.search(r"(\d{1,2}):(\d{2})", task_text)

    remind_at_str = None
    if time_match:
        hour = int(time_match.group(1))
        minute = int(time_match.group(2))

        target_time = now.replace(hour=hour, minute=minute, second=0)
        if target_time < now:
            target_time += timedelta(days=1)

        remind_at_str = target_time.strftime("%Y-%m-%d %H:%M")

    tasks_data = load_tasks()

    if user_id not in tasks_data or isinstance(tasks_data[user_id], list):
        existing_tasks = (
            tasks_data[user_id]
            if user_id in tasks_data and isinstance(tasks_data[user_id], list)
            else []
        )
        tasks_data[user_id] = {"name": user_full_name, "tasks": existing_tasks}

    tasks_data[user_id]["name"] = user_full_name

    task_obj = {"text": task_text, "remind_at": remind_at_str}
    tasks_data[user_id]["tasks"].append(task_obj)
    save_tasks(tasks_data)

    msg = f"📌 **Vazifa saqlandi:** {clean_markdown(task_text)}\n"
    if remind_at_str:
        msg += f"⏰ **Eslatma vaqti o'rnatildi:** {remind_at_str}"
    else:
        msg += "⚠️ (Vaqt topilmadi. Soatni HH:MM formatida yozing)"

    safe_send_message(
        message.chat.id, msg, parse_mode="Markdown", reply_markup=main_keyboard()
    )
